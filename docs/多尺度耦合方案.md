# 多尺度耦合方案：颗粒应力到宏观Jellyroll网格

## 问题分析

### 尺度差异

```
微观尺度（颗粒）                宏观尺度（Jellyroll网格）
┌─────────────┐                ┌──────────────────────────┐
│ 颗粒直径:    │                │ 网格单元尺寸:             │
│ ~10 μm      │   ====>        │ ~1 mm (r方向)            │
│             │   耦合方法?     │ ~几mm (θ方向)            │
│ 扩散应力     │                │ 宏观应力场                │
└─────────────┘                └──────────────────────────┘
    10^-5 m                         10^-3 m
    ↑                                ↑
  微观力学                         连续介质力学
```

**尺度比：** ~100倍

## 解决方案：多尺度均匀化

### 方法1：体积平均法（Volume Averaging）⭐⭐⭐ 推荐

#### 理论基础

宏观应力 = 代表性体积元（RVE）内颗粒应力的体积平均

```
σ_macro = (1/V_RVE) ∫∫∫ σ_particle dV

对于多颗粒系统：
σ_macro = Σ(V_i / V_total) · σ_i

其中：
- V_i = 单个颗粒体积
- V_total = 电极体积（颗粒+孔隙+粘结剂）
- σ_i = 第i个颗粒的应力
```

#### 体积分数修正

```
ε_active = 活性材料体积分数（通常 0.6-0.8）

有效宏观应力：
σ_macro_eff = ε_active · σ_particle_avg

物理意义：
- 只有活性材料承载扩散应力
- 孔隙和粘结剂不贡献扩散应力
- 但它们分担总体变形
```

#### 代码实现

```julia
"""
从颗粒尺度应力均匀化到宏观单元应力
"""
function homogenize_particle_stress_to_element(
    case::Case,
    variables::Dict,
    mesh_th::Mesh
)
    param = case.param_dim
    ne_th = size(mesh_th.element, 1)
    
    # 提取颗粒应力
    if haskey(variables, "negative particle surface tangential stress")
        σ_particle_n = variables["negative particle surface tangential stress"]
        σ_particle_p = variables["positive particle surface tangential stress"]
    else
        return zeros(Float64, ne_th)
    end
    
    # 材料参数
    ε_active_n = hasproperty(param.NE, :epsilon_s) ? param.NE.epsilon_s : 0.65
    ε_active_p = hasproperty(param.PE, :epsilon_s) ? param.PE.epsilon_s : 0.60
    
    t_n = param.NE.thickness
    t_p = param.PE.thickness
    t_total = t_n + t_p
    
    # 厚度加权平均（负极和正极）
    if isa(σ_particle_n, Number)
        # SPM/SPMe: 标量应力
        σ_avg_n = σ_particle_n * ε_active_n
        σ_avg_p = σ_particle_p * ε_active_p
        
        # 厚度加权
        σ_macro = (σ_avg_n * t_n + σ_avg_p * t_p) / t_total
        
        # 分配到所有单元（均匀）
        σ_element = fill(σ_macro, ne_th)
        
    else
        # P2D: 向量应力（沿x轴分布）
        # 需要进一步处理：从1D x轴映射到2D (r,θ)网格
        σ_element = map_1D_to_2D_jellyroll(σ_particle_n, σ_particle_p, mesh_th, param)
    end
    
    return σ_element
end

"""
将1D轴向应力场映射到2D Jellyroll网格
"""
function map_1D_to_2D_jellyroll(
    σ_n_1D::Vector{Float64},
    σ_p_1D::Vector{Float64},
    mesh_th::Mesh,
    param
)
    ne_th = size(mesh_th.element, 1)
    σ_element = zeros(Float64, ne_th)
    
    # 获取单元中心坐标
    centers = jellyroll_element_centers(mesh_th)
    r_centers = sqrt.(centers[:,1].^2 .+ centers[:,2].^2)
    
    # Jellyroll螺旋参数
    Rin = param.cell.Rin
    Rout = param.cell.Rout
    t_layer = param.NE.thickness + param.SE.thickness + param.PE.thickness
    
    # 方法1：基于径向位置的线性插值
    # 假设1D应力沿径向线性分布
    n_x = length(σ_n_1D)
    x_positions = range(0, t_layer, length=n_x)
    
    # 体积分数
    ε_active_n = hasproperty(param.NE, :epsilon_s) ? param.NE.epsilon_s : 0.65
    ε_active_p = hasproperty(param.PE, :epsilon_s) ? param.PE.epsilon_s : 0.60
    
    # 厚度加权
    σ_weighted = (σ_n_1D .* ε_active_n .* param.NE.thickness .+ 
                  σ_p_1D .* ε_active_p .* param.PE.thickness) ./ 
                 (param.NE.thickness + param.PE.thickness)
    
    # 插值到每个单元
    for e in 1:ne_th
        # 方法：使用径向位置作为代理
        # 简化假设：每圈螺旋的应力相同
        r_e = r_centers[e]
        
        # 归一化径向位置 [0,1]
        r_norm = (r_e - Rin) / (Rout - Rin)
        
        # 插值到1D应力场
        idx = max(1, min(n_x, Int(ceil(r_norm * n_x))))
        σ_element[e] = σ_weighted[idx]
    end
    
    return σ_element
end
```

### 方法2：等效应变源项法（Equivalent Strain Source）⭐⭐

#### 理论

将颗粒尺度的应力转化为宏观的**等效体积应变**，作为热膨胀类比

```
颗粒膨胀位移：
u_surf = (Ω·R·c̄)/3

等效宏观应变：
ε_vol = 3·u_surf/R = Ω·c̄

等效"热膨胀系数"：
α_eff = ε_vol / Δc = Ω

宏观应力（通过热应力公式）：
σ_macro = E_eff·α_eff·Δc/(1-ν_eff)
```

#### 代码实现

```julia
"""
等效应变源项法
"""
function diffusion_stress_as_thermal_analogy(case::Case, variables::Dict)
    if !haskey(variables, "negative particle surface displacement")
        return variables
    end
    
    param = case.param_dim
    mesh_th = case.mesh["thermal2D"]
    ne_th = size(mesh_th.element, 1)
    
    # 颗粒表面位移
    u_n = variables["negative particle surface displacement"]
    u_p = variables["positive particle surface displacement"]
    
    # 等效体积应变
    rs_n = param.NE.rs
    rs_p = param.PE.rs
    ε_vol_n = 3 * u_n / rs_n  # 如果u是标量
    ε_vol_p = 3 * u_p / rs_p
    
    # 材料参数
    E_n = hasproperty(param.NE, :E) ? param.NE.E : 1e9
    E_p = hasproperty(param.PE, :E) ? param.PE.E : 1e9
    ν_n = hasproperty(param.NE, :nu) ? param.NE.nu : 0.3
    ν_p = hasproperty(param.PE, :nu) ? param.PE.nu : 0.3
    
    # 等效宏观应力
    σ_n_macro = E_n * ε_vol_n / (1 - ν_n)
    σ_p_macro = E_p * ε_vol_p / (1 - ν_p)
    
    # 厚度加权
    t_n = param.NE.thickness
    t_p = param.PE.thickness
    σ_macro = (σ_n_macro * t_n + σ_p_macro * t_p) / (t_n + t_p)
    
    # 分配到所有单元
    if isa(σ_macro, Number)
        σ_element = fill(σ_macro, ne_th)
    else
        # 处理向量情况
        σ_element = map_1D_to_2D_jellyroll_simple(σ_macro, mesh_th)
    end
    
    variables["thermal2D element diffusion stress (homogenized)"] = σ_element
    
    return variables
end
```

### 方法3：基于层权重的精细映射（Layer-Weight Based）⭐⭐⭐ 最精确

#### 理论

利用Jellyroll网格已有的`layer_weights`（负极/隔膜/正极的体积分数）

```
每个单元的应力 = 负极应力·f_neg + 正极应力·f_pos

其中：
- f_neg = 单元内负极层的体积分数
- f_pos = 单元内正极层的体积分数
```

#### 代码实现

```julia
"""
基于layer_weights的精细映射（推荐用于Jellyroll）
"""
function homogenize_with_layer_weights(case::Case, variables::Dict)
    if !haskey(case.mesh, "thermal2D")
        return variables
    end
    
    mesh_th = case.mesh["thermal2D"]
    ne_th = size(mesh_th.element, 1)
    param = case.param_dim
    
    # 获取layer_weights（如果可用）
    fks = nothing
    if haskey(variables, "thermal2D layer_weights")
        fks = variables["thermal2D layer_weights"]
    else
        # 尝试从mesh中获取或重新计算
        try
            fks = jellyroll_get_layer_weights(mesh_th)
            if fks === nothing
                fks = jellyroll_element_layer_weights(mesh_th, param; 
                                                      nsamples_per_dim=4, 
                                                      logic=:spiral)
            end
        catch e
            @warn "无法获取layer_weights，使用均匀分布" exception=(e, catch_backtrace())
            fks = nothing
        end
    end
    
    # 提取颗粒应力
    σ_particle_n = variables["negative particle surface tangential stress"]
    σ_particle_p = variables["positive particle surface tangential stress"]
    
    # 体积分数
    ε_active_n = hasproperty(param.NE, :epsilon_s) ? param.NE.epsilon_s : 0.65
    ε_active_p = hasproperty(param.PE, :epsilon_s) ? param.PE.epsilon_s : 0.60
    
    # 有效应力（考虑孔隙）
    σ_eff_n = isa(σ_particle_n, Number) ? σ_particle_n * ε_active_n : σ_particle_n .* ε_active_n
    σ_eff_p = isa(σ_particle_p, Number) ? σ_particle_p * ε_active_p : σ_particle_p .* ε_active_p
    
    # 分配到单元
    σ_element = zeros(Float64, ne_th)
    
    if fks !== nothing && size(fks, 2) >= 3
        # 使用layer_weights精细映射
        @inbounds for e in 1:ne_th
            f_neg = fks[e, 1]  # 负极层权重
            f_sep = fks[e, 2]  # 隔膜层权重
            f_pos = fks[e, 3]  # 正极层权重
            
            # 加权平均（隔膜不贡献应力）
            if isa(σ_eff_n, Number)
                σ_element[e] = f_neg * σ_eff_n + f_pos * σ_eff_p
            else
                # P2D向量情况：需要映射到单元
                # 简化：使用平均值
                σ_element[e] = f_neg * mean(σ_eff_n) + f_pos * mean(σ_eff_p)
            end
        end
    else
        # 退化到简单厚度加权
        t_n = param.NE.thickness
        t_p = param.PE.thickness
        t_total = t_n + t_p
        
        if isa(σ_eff_n, Number)
            σ_macro = (σ_eff_n * t_n + σ_eff_p * t_p) / t_total
            σ_element .= σ_macro
        else
            σ_macro = (mean(σ_eff_n) * t_n + mean(σ_eff_p) * t_p) / t_total
            σ_element .= σ_macro
        end
    end
    
    variables["thermal2D element diffusion stress (homogenized)"] = σ_element
    
    # 叠加热应力
    if haskey(variables, "thermal2D element thermal stress")
        σ_thermal = variables["thermal2D element thermal stress"]
        variables["thermal2D element total stress"] = σ_thermal .+ σ_element
    end
    
    return variables
end
```

```julia
"""
    homogenize_particle_stress_to_2D(case::Case, variables::Dict)

多尺度均匀化：将颗粒尺度应力映射到宏观2D网格

# 方法
1. 提取颗粒表面应力（微米尺度）
2. 应用体积分数修正（活性材料占比）
3. 使用layer_weights进行精细映射（如果可用）
4. 厚度加权平均（负极+正极）
5. 输出宏观单元应力场

# 理论基础
σ_macro = Σ(f_layer · ε_active · σ_particle)

# 参数
- `case::Case` - JuBat案例对象
- `variables::Dict` - 包含颗粒应力的变量字典

# 返回
- 更新后的`variables`，新增：
  - `"thermal2D element diffusion stress (homogenized)"` - 均匀化扩散应力 [Pa]
  - `"thermal2D element total stress"` - 总应力（热+扩散） [Pa]
"""
function homogenize_particle_stress_to_2D(case::Case, variables::Dict)
    # 检查是否有2D网格
    if !haskey(case.mesh, "thermal2D")
        return variables
    end
    
    # 检查是否有颗粒应力
    if !haskey(variables, "negative particle surface tangential stress")
        return variables
    end
    
    mesh_th = case.mesh["thermal2D"]
    ne_th = size(mesh_th.element, 1)
    param = case.param_dim
    
    # 提取颗粒应力
    σ_particle_n = variables["negative particle surface tangential stress"]
    σ_particle_p = variables["positive particle surface tangential stress"]
    
    # 体积分数（活性材料）
    ε_active_n = hasproperty(param.NE, :epsilon_s) ? param.NE.epsilon_s : 0.65
    ε_active_p = hasproperty(param.PE, :epsilon_s) ? param.PE.epsilon_s : 0.60
    
    # 有效应力
    σ_eff_n = isa(σ_particle_n, Number) ? σ_particle_n * ε_active_n : σ_particle_n .* ε_active_n
    σ_eff_p = isa(σ_particle_p, Number) ? σ_particle_p * ε_active_p : σ_particle_p .* ε_active_p
    
    # 获取layer_weights（优先使用）
    fks = nothing
    if haskey(variables, "thermal2D layer_weights")
        fks = variables["thermal2D layer_weights"]
    end
    
    # 分配到单元
    σ_element = zeros(Float64, ne_th)
    
    if fks !== nothing && size(fks, 2) >= 3
        # 方法1：精细映射（使用layer_weights）
        @inbounds for e in 1:ne_th
            f_neg = fks[e, 1]
            f_pos = fks[e, 3]
            
            if isa(σ_eff_n, Number)
                σ_element[e] = f_neg * σ_eff_n + f_pos * σ_eff_p
            else
                # P2D: 需要额外的空间映射逻辑
                # 简化：使用平均值
                σ_element[e] = f_neg * mean(σ_eff_n) + f_pos * mean(σ_eff_p)
            end
        end
        
    else
        # 方法2：简单厚度加权
        t_n = param.NE.thickness
        t_p = param.PE.thickness
        t_total = t_n + t_p
        
        if isa(σ_eff_n, Number)
            σ_macro = (σ_eff_n * t_n + σ_eff_p * t_p) / t_total
            σ_element .= σ_macro
        else
            σ_macro = (mean(σ_eff_n) * t_n + mean(σ_eff_p) * t_p) / t_total
            σ_element .= σ_macro
        end
    end
    
    # 存储结果
    variables["thermal2D element diffusion stress (homogenized)"] = σ_element
    
    # 叠加热应力（如果已计算）
    if haskey(variables, "thermal2D element thermal stress")
        σ_thermal = variables["thermal2D element thermal stress"]
        variables["thermal2D element total stress"] = σ_thermal .+ σ_element
    else
        variables["thermal2D element total stress"] = σ_element
    end
    
    return variables
end
```
