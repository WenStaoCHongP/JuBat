# 结构体重构方案：替代 Dict{String,Any}

> 日期: 2026-04-01
> 目标：消除 `multi_spme_layout::Dict{String,Any}` 的类型不安全问题

---

## 1. 当前问题诊断

### 1.1 `multi_spme_layout` 存储的三类数据

| 类别 | 键 | 设置位置 | 变化频率 | 性质 |
|------|-----|---------|---------|------|
| **布局索引** | `ne`, `n_chem`, `nT`, `n_total`, `chem_range`, `thermal_range` | Initialisation / CycleSolver | 一次 | 纯数值 / Range |
| **网格几何** | `interface_pairs`, `element_layer`, `is_inner_layer`, `czm_element_map`, `inner_nodes`, `outer_nodes` | Jellyrollmodel | 一次 | 几何拓扑 |
| **运行时状态** | `thermal_variables`, `thermal_update_fn`, `thermal_record`, `czm_mesh` | Solve / CycleSolver | 每步 | 可变对象 |

### 1.2 布局构建代码 4 处重复

相同 6 行代码出现在：

| 位置 | 文件:行号 |
|------|----------|
| `ModelInitialisation_MultiSPMe` | Initialisation.jl:110-116 |
| `Solve` 内 multi-SPMe 检测 | Solve.jl:104-109 |
| `CallModel` 自动初始化 | Solve.jl:598-603 |
| `_ensure_multi_spme_layout!` | CycleSolver.jl:708-713 |

### 1.3 `_ensure_multi_spme_layout!` 的存在本身就是 bug 风险

该函数之所以需要，是因为 `Dict` 可能为空（状态从上阶段传递时布局信息丢失）。如果使用不可变 struct，一旦设置就不会丢失。

## 2. 新增结构体定义

### 2.1 `MultiSPMeLayout`（替代布局索引键）

```julia
# CouplingState.jl

"""
多SPMe状态向量的布局索引。
初始化后不可变——所有字段均为 const。
"""
struct MultiSPMeLayout
    ne::Int                        # 热单元数
    n_chem::Int                    # 每单元电化学 DOF 数 = Nrn + Nrp + Nel
    nT::Int                        # 热节点 DOF 数
    n_total::Int                   # 全局状态向量总长 = ne*n_chem + nT
    chem_range::UnitRange{Int}     # 1:(ne*n_chem)
    thermal_range::UnitRange{Int}  # (ne*n_chem+1):(ne*n_chem+nT)
end

# 便捷构造器：自动计算 range 和 n_total
function MultiSPMeLayout(ne::Int, n_chem::Int, nT::Int)
    MultiSPMeLayout(
        ne, n_chem, nT,
        ne * n_chem + nT,
        1:(ne * n_chem),
        (ne * n_chem + 1):(ne * n_chem + nT)
    )
end
```

**替代效果**：4 处重复的 6 行代码 → 4 处 1 行调用 `MultiSPMeLayout(ne, n_chem, nT)`

### 2.2 `MeshGeometry`（替代网格几何键）

```julia
# CouplingState.jl

"""
Jellyroll 网格的几何拓扑信息。
构建后不可变。
"""
struct MeshGeometry
    element_layer::Vector{Int}              # 每个单元的层类型 (1=NE, 2=SP, 3=PE, 4=NCC, 5=PCC)
    is_inner_layer::BitVector               # 是否为内层
    layer_weights::Matrix{Float64}          # ne × 5 层面积权重矩阵
    interface_pairs::Vector{Tuple{Int,Int}} # CZM 界面配对 (top_elem, bot_elem)
    czm_element_map::Dict{Int,Int}          # 热单元号 → CZM 单元号
    inner_nodes::Vector{Int}                # 内边界节点索引
    outer_nodes::Vector{Int}                # 外边界节点索引
end
```

## 3. Case 结构体重构

### 3.1 重构前

```julia
# SetCase.jl (当前)
mutable struct Case
    param_dim::Params
    param::Params
    opt::Option
    mesh::Dict{String, Mesh}
    index::Dict{String, Union{Array{Int64}, Int64}}
    multi_spme_layout::Dict{String,Any}    # ← 问题根源
end
```

### 3.2 重构后

```julia
# SetCase.jl
mutable struct Case
    # ---- main 分支原字段，不动 ----
    param_dim::Params
    param::Params
    opt::Option
    mesh::Dict{String, Mesh}
    index::Dict{String, Union{Array{Int64}, Int64}}

    # ---- 替代 multi_spme_layout ----
    layout::Union{Nothing, MultiSPMeLayout}   # 布局索引（初始化后不变）
    geometry::Union{Nothing, MeshGeometry}     # 几何拓扑（构建后不变）
    czm_mesh::Union{Nothing, CohesiveMesh}     # CZM 网格（演化但类型明确）
end

# 向后兼容构造器（main 分支标准路径使用）
function Case(param_dim, param, opt, mesh, index)
    Case(param_dim, param, opt, mesh, index, nothing, nothing, nothing)
end
```

### 4.3 `Initialisation.jl` — 辅助函数简化

```julia
# 旧 MultiSPMe_extract_element_state:
function MultiSPMe_extract_element_state(y::Array{Float64}, e::Int, case::Case)
    y_vec = vec(y)
    layout = case.multi_spme_layout
    ne = layout["ne"]
    n_chem = layout["n_chem"]
    offset = (e - 1) * n_chem
    yt_e = y_vec[(offset + 1):(offset + n_chem)]
    return yt_e
end

# 新:
function extract_element_state(y::AbstractVector, e::Int, layout::MultiSPMeLayout)
    offset = (e - 1) * layout.n_chem
    return y[(offset + 1):(offset + layout.n_chem)]
end
```

```julia
# 旧 MultiSPMe_get_thermal_dofs:
function MultiSPMe_get_thermal_dofs(y::Array{Float64}, case::Case)
    y_vec = vec(y)
    layout = case.multi_spme_layout
    thermal_range = layout["thermal_range"]
    return y_vec[thermal_range]
end

# 新:
function get_thermal_dofs(y::AbstractVector, layout::MultiSPMeLayout)
    return y[layout.thermal_range]
end
```

```julia
# 旧 MultiSPMe_update_state (35行):
function MultiSPMe_update_state(y, case; element_index=nothing, element_state=nothing, thermal_nodes=nothing)
    # ... 35 行，含手动 Dict 检查和 offset 计算
end

# 新:
function update_state(y::AbstractVector, layout::MultiSPMeLayout;
                      element_index::Union{Nothing,Int}=nothing,
                      element_state::Union{Nothing,Vector{Float64}}=nothing,
                      thermal_nodes::Union{Nothing,Vector{Float64}}=nothing)
    y_new = copy(y)
    if element_index !== nothing
        offset = (element_index - 1) * layout.n_chem
        y_new[(offset + 1):(offset + layout.n_chem)] .= element_state
    end
    if thermal_nodes !== nothing
        y_new[layout.thermal_range] .= thermal_nodes
    end
    return y_new
end
```

## 5. 消除的问题汇总

| 问题 | 消除方式 |
|------|---------|
| 4 处重复的布局构建 (各 6 行) | `MultiSPMeLayout` 构造器自动计算 |
| `Dict{String,Any}` 无类型安全 | struct 字段有明确类型 |
| 拼写错误只在运行时暴露 | 编译期检查 |
| `_ensure_multi_spme_layout!` 函数 | `layout === nothing` 一行判断 |
| 几何数据与运行时状态混存 | `MeshGeometry` / `CohesiveMesh` 分离 |
