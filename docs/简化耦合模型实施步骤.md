```julia
function CallModel_SimpleCoupling(case, y, t; jacobi="update")
    layout = case.simple_coupling_layout
    n_chem = layout["n_chem"]
    nT = layout["nT"]
    
    # 1. 提取状态
    y_chem = y[1:n_chem]
    y_T = y[n_chem+1:end]
    
    # 2. 计算平均温度反馈
    T_avg = mean(y_T) * case.param.scale.T0
    case.param.T0 = T_avg
    
    # 3. 求解SPMe
    M_chem, K_chem, F_chem = solve_single_spme(case, y_chem, t)
    
    # 4. 计算总内热源并分配
    Q_total = compute_total_heat_source(case, y_chem)
    q_volumetric = distribute_heat_source(case, Q_total)
    
    # 5. 求解热传导
    M_T, K_T, F_T = solve_thermal2D(case, y_T, q_volumetric)
    
    # 6. 组装系统矩阵
    M = block_diagonal(M_chem, M_T)
    K = block_diagonal(K_chem, K_T)
    F = vcat(F_chem, F_T)
    
    # 7. 组装变量
    variables = Dict(
        "voltage" => ...,
        "current" => ...,
        "thermal2D temperature" => y_T * case.param.scale.T0,
        "average temperature" => T_avg,
        "total heat source" => Q_total
    )
    
    return M, K, F, variables, y
end
```

## 关键函数实现

### 热源计算与分配
```julia
function compute_total_heat_source(case, y_chem)
    # 从SPMe状态计算各项损耗
    η_n, η_p = compute_overpotentials(case, y_chem)
    i_n, i_p = compute_interfacial_currents(case, y_chem)
    
    # 反应热（过电位）
    Q_rxn = compute_reaction_heat(η_n, η_p, i_n, i_p, case.param)
    
    # 欧姆热
    Q_ohm = compute_ohmic_heat(case, y_chem)
    
    # 熵热
    Q_rev = compute_reversible_heat(case, y_chem)
    
    Q_total = Q_rxn + Q_ohm + Q_rev  # 单位：W
    return Q_total
end

function distribute_heat_source(case, Q_total)
    # 计算总体积
    mesh = case.mesh["thermal2D"]
    ne = size(mesh.element, 1)
    
    V_total = 0.0
    for e in 1:ne
        V_total += compute_element_volume(mesh, e)
    end
    
    # 均匀体积热源密度（W/m³）
    q_volumetric = Q_total / V_total
    
    # 返回每个单元的热源（均匀）
    return fill(q_volumetric, ne)
end
```

### 单元体积计算
```julia
function compute_element_volume(mesh, e)
    # 对于2D网格，需要×高度H
    nodes = mesh.element[e, :]
    coords = mesh.node[nodes, :]
    
    # Q4单元面积（使用高斯积分）
    A = 0.0
    for g in 1:mesh.gs.ngs
        if mesh.gs.ele[g] == e
            A += mesh.gs.weight[g] * mesh.gs.detJ[g]
        end
    end
    
    # 体积 = 面积 × 高度
    H = case.param.cell.height
    V = A * H
    
    return V
end
```
